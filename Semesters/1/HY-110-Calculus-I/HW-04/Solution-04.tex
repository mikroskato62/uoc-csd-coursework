\[
\textbf{Αρχικά, ξέρουμε:} \hspace{0.321cm}
\log _a (b) = \frac{\ln (b)}{\ln (a)} \hspace{0.12345cm} \Rightarrow \hspace{0.12345cm} \log _2(x)=\frac{\ln (x)}{\ln (2)} \hspace{25cm}
\]

\[
\textbf{Άρα:} \hspace{0.321cm}
\lim_{x \to \infty} \left( \frac{\ln (x+1)}{\log_2 (x)} \right) = 
\lim _{x \rightarrow \infty}\left(\frac{\ln (x+1)}{\frac{\ln (x)}{\ln (2)}}\right)=\ln (2) \hspace{0.1cm} \cdot \hspace{0.1cm} \lim _{x \rightarrow \infty}\left(\frac{\ln (x+1)}{\ln (x)}\right)=\ln (2) \hspace{0.1cm} \cdot \hspace{0.1cm} \lim _{x \rightarrow \infty}\left(\frac{\ln \left(x\left(1+\frac{1}{x}\right)\right)}{\ln (x)}\right) \hspace{0.1cm} \Rightarrow \hspace{25cm}
\]

\[
\Rightarrow \hspace{0.2cm} \ln (2) \hspace{0.1cm} \cdot \hspace{0.1cm} \lim _{x \rightarrow \infty}\left(\frac{\ln (x)+\ln \left(1+\frac{1}{x}\right)}{\ln (x)}\right)=\ln (2) \lim _{x \rightarrow \infty}\left(1+\frac{\ln \left(1+\frac{1}{x}\right)}{\ln (x)}\right) = \ln (2) \hspace{0.1cm} \cdot \hspace{0.1cm} \left[\lim _{x \rightarrow \infty}(1)+\lim _{x \rightarrow \infty}\left(\frac{\ln \left(1+\frac{1}{x}\right)}{\ln (x)}\right)\right] \hspace{0.1cm} \Rightarrow \hspace{25cm}
\]

\[
\Rightarrow \hspace{0.2cm} \ln (2) \hspace{0.1cm}\cdot\hspace{0.1cm}\left[1+\frac{\ln (1 + 0)}{\ln (\infty)}= 1 + \frac{0}{\ln (\infty)}\right] 
=\ln (2) \hspace{0.1cm}\cdot\hspace{0.1cm}[1+0]=\ln (2) \hspace{0.1cm} \hspace{25cm}
\]

\[
\textbf{Τελικά:} \hspace{0.321cm}
\lim_{x \to \infty} \left( \frac{\ln (x+1)}{\log_2 (x)} \right) 
\hspace{0.1cm} = \hspace{0.1cm} \boxed{\ln2} \hspace{25cm}

\textbf{Αφού:} \hspace{0.321cm} x \hspace{0.1cm}\textbf{ε}\hspace{0.1cm} [0, \hspace{0.1cm}2] \hspace{0.1cm}\textbf{τότε}\hspace{0.1cm} (x-1) \hspace{0.1cm}\textbf{ε} \hspace{0.1cm}[-1,\hspace{0.1cm}1] \hspace{0.1cm}\textbf{οπότε:} \hspace{0.25cm}
\int_0^2 \frac{1}{\sqrt{|x-1|}} d x=\int_0^1 \frac{1}{\sqrt{|x-1|}} d x+\int_{01}^2 \frac{1}{\sqrt{|x-1|}} d x=I_1+I_2
\hspace{25cm}
\]

\[
\textbf{·} \hspace{0.321cm} I_1: \hspace{0.321cm} 
\int_0^1 \frac{1}{\sqrt{|x-1|}} d x \hspace{0.35cm}\textbf{και}\hspace{0.35cm}
x-1 \leqslant 0 \quad \forall_{x \in}[0,1] \hspace{0.35cm}\textbf{άρα}\hspace{0.35cm} |x-1|=-(x-1)=1-x
\hspace{0.35cm}\textbf{και} \hspace{0.35cm} \int_0^1 \frac{1}{\sqrt{1 - x}} dx
\hspace{25cm}
\]

\[
\textbf{Θέτω:} \hspace{0.321cm}
u = 1 - x \hspace{0.35cm}\textbf{και} \hspace{0.35cm} u' = \frac{du}{dx} = -1 \Rightarrow dx = -du
\hspace{0.35cm}\textbf{και} \hspace{0.35cm} x = 0 \Rightarrow u = 1 \hspace{0.35cm}\textbf{και} \hspace{0.35cm}
x = 1 \Rightarrow u = 0 \hspace{0.35cm}\textbf{οπότε:} \hspace{0.35cm}
\hspace{25cm}
\]

\[
I_1=
\int_1^0 \frac{-1}{\sqrt{u}} d x
\hspace{0.35cm}\textbf{ή} \hspace{0.35cm}
I_1=\int_0^1 \frac{1}{\sqrt{u}} d u=[2 \sqrt{u}]_0^1=2\left(\sqrt{1}-\sqrt{0}\right)=2
\]

\[
\textbf{·} \hspace{0.321cm} I_2: \hspace{0.321cm} 
\int_1^{\varepsilon} \frac{1}{\sqrt{|x-1|}} d x 
\hspace{0.35cm}\textbf{και} \hspace{0.35cm} x-1 \geqslant 0 \quad \forall x \in[1,2]
\hspace{0.35cm}\textbf{άρα} \hspace{0.35cm} |x-1| = x-1 \hspace{0.35cm}\textbf{και} \hspace{0.35cm}
\int_1^2 \frac{1}{\sqrt{x-1}} d x
\hspace{25cm}
\]

\[
\textbf{Θέτω:} \hspace{0.321cm}
u = x-1 \hspace{0.35cm}\textbf{και} \hspace{0.35cm} u' = \frac{du}{dx} = 1 \Rightarrow dx = du
\hspace{0.35cm}\textbf{και} \hspace{0.35cm} x =1 \Rightarrow u = 0 \hspace{0.35cm}\textbf{και} \hspace{0.35cm}
x = 2 \Rightarrow u = 1 \hspace{0.35cm}\textbf{οπότε:} \hspace{0.35cm}
\hspace{25cm}
\]

\[
I_2=\int_0^1 \frac{1}{\sqrt{u}} d u=[2 \sqrt{u}]_0^1=2(\sqrt{1}-\sqrt{0})=2\cdot1=2
\]

\vspace{0.75cm}

\[
\textbf{Τελικά:} \hspace{0.321cm}
I=I_1+I_2=2+2=4 \hspace{0.35cm}\textbf{και έτσι:} \hspace{0.35cm}
\int_0^2 \frac{d x}{\sqrt{| x-1 |}}= \boxed{4}
\hspace{25cm}
\]

\hspace{25cm}

\[
\textbf{Έστω} \hspace{0.321cm}
f(x)=x^2 e^{-x}(x \in \mathbb{R}) \hspace{0.35cm}\textbf{τότε} \hspace{0.35cm} f(x)
\hspace{0.35cm}\textbf{συνεχής στο} \hspace{0.35cm} [0, \infty] \hspace{0.35cm}\textbf{ως πράξεις συνεχών.}
\hspace{25cm}

\[
\textbf{Έχουμε:} \hspace{0.321cm}
I=\lim _{b \rightarrow \infty} \int_0^b x^2 e^{-x} d x \hspace{0.123cm}\Rightarrow\hspace{0.123cm} I=\lim _{b \rightarrow \infty}(\omega)
\hspace{25cm}

\[
\omega \hspace{0.123cm} = \hspace{0.123cm}
\int_0^b x^2 e^{-x} d x=\int_0^b x^2\left(-e^{-x}\right)^{\prime} d x=\left[-x^2 e^{-x}\right]_0^e+\int_a^e 2 x e^{-x} d x=
[-x^2 e^{-x}]+2 \int_0^e x \cdot\left(-e^{-x}\right)^{\prime} d x

\[
=[-x^2 e^{-x}]+2\left(\left[-x e^{-x}\right]_0^b+\int_0^b e^{-x} d x\right)=
[-x^2 e^{-x}]+2\left([-x e^{-x}]+\left[-e^{-x}\right]_0^b\right)=

\]

\[
\textbf{ή αλλιώς: } \hspace{0.321cm}
\omega \hspace{0.123cm} = \hspace{0.123cm}

\hspace{25cm}

\[
\textbf{Άρα:} \hspace{0.321cm}
I=\lim _{b \rightarrow \infty}\left(\left[-x^2 e^{-x}-2 x e^{-x}-2 e^{-x}\right]_0^b\right)
=(-(\infty) \cdot 0-(\infty) \cdot 0 \cdot 0)-(0-0 - 2) = +2
\hspace{25cm}
\]

\[
\textbf{Τελικά:} \hspace{0.321cm}
\int_0^{\infty} x^2 e^{-x} d x= \boxed{2} \hspace{25cm}
